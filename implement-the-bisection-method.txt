def square_root_bisection(num:float,tol=1e-6,max_it=100):
    if num<0:
        raise ValueError("Square root of negative number is not defined in real numbers")
    elif num==0 or num==1:
        print(f"The square root of {num} is {num}")
        return num
        
    low,high=(0,num) if num>1 else (0,1)
    for _ in range(max_it):
        mid=(low+high)/2
        s=mid*mid
        if abs(s-num)<=tol and high-low<=tol:
            print(f"The square root of {num} is approximately {mid}")
            return mid
        if s<num:
            low=mid
        else:
            high=mid
    print(f"Failed to converge within {max_it} iterations")
    return None






square_root_bisection(0.001, 1e-7, 50)
square_root_bisection(0.001, 1e-7, 50)
square_root_bisection(81, 1e-3, 50)
