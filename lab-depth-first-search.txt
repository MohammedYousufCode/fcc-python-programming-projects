def dfs(adj_mat,start_node):
    n=len(adj_mat)
    visited=[False]*n
    stack=[start_node]
    result=[]
    while stack:
        node=stack.pop()
        if not visited[node]:
            visited[node]=True
            result.append(node)
            for neighbor in range(n-1,-1,-1):
                if adj_mat[node][neighbor]==1 and not visited[neighbor]:
                    stack.append(neighbor)
    return result
