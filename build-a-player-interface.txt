from abc import ABC, abstractmethod
import random

class Player(ABC):
    def __init__(self):
        self.moves = []
        self.position = (0, 0)
        self.path = [self.position]

    def make_move(self):
        dx, dy = random.choice(self.moves)
        x, y = self.position
        self.position = (x + dx, y + dy)
        self.path.append(self.position)
        return self.position

    @abstractmethod
    def level_up(self):
        pass

class Pawn(Player):
    def __init__(self):
        super().__init__()
        # Up, Down, Left, Right
        self.moves = [(0, 1), (0, -1), (-1, 0), (1, 0)]

    def level_up(self):
        # Add diagonal moves: Up-Left, Up-Right, Down-Left, Down-Right
        self.moves += [(-1, 1), (1, 1), (-1, -1), (1, -1)]
