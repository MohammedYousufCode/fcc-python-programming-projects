def quick_sort(array):
    if len(array)<=1:
        return array
    pivot=array[len(array)-1]
    left=[]
    mid=[]
    right=[]
    for a in array:
        if a<pivot:
            left.append(a)
        elif a==pivot:
            mid.append(a)
        elif a>pivot:
            right.append(a)
    sorted_left=quick_sort(left)
    sorted_right=quick_sort(right)
    array=sorted_left+mid+sorted_right
    return array
array=[2,4,1,3,5,6,5,7,5,6,9,8,7,9]
print(quick_sort(array))
